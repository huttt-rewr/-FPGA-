// Per-pixel effects: passthrough / invert / Sobel edge_detected / thermal false-color.
// Fixed 3-cycle latency on all modes; vs/de/rgb delayed together so control and
// pixels stay aligned. Sobel uses a causal 3x3 window (2 line buffers + 3-tap
// column shifts), so its output is shifted +1 pixel/+1 line vs the input --
// the standard, visually-invisible cost of a streaming 3x3 window.
//
// Line buffers are async-read (combinational L1[x]) + sync-write, which maps to
// distributed RAM (LUT). If the tool infers block RAM instead, two 1024x8
// buffers still cost only 2 BRAM9K. DEPTH must be >= the widest active line
// (640 for VGA, 800 for SVGA); 1024 covers both.
//
// Modes (mode_sel):
//   00 passthrough   01 invert   10 Sobel (white edges on black)   11 thermal
module pixel_effects #(
    parameter DEPTH = 1024,
    parameter [7:0] EDGE_TH = 8'd80
)(
    input clk, rst,                      // rst active-high (matches project)
    input vs, de,
    input [9:0] x,
    input [23:0] rgb,                    // RGB888
    input [1:0] mode_sel,
    output reg out_vs, output reg out_de,
    output reg [23:0] out_rgb
);
    // Integer luma, shared by Sobel and thermal.
    wire [7:0] gray = (77*rgb[23:16] + 150*rgb[15:8] + 29*rgb[7:0]) >> 8;

    // Two previous rows. Async read gives the OLD value before the sync write,
    // so read-while-write on the same address is safe (read-first semantics).
    reg [7:0] L1 [0:DEPTH-1];            // row y-1
    reg [7:0] L2 [0:DEPTH-1];            // row y-2
    wire [7:0] l1 = L1[x];
    wire [7:0] l2 = L2[x];
    // Zero at time 0 so the first frame's boundary reads are clean in
    // simulation. Synthesis ignores this (RAM init is don't-care until the
    // first full frame overwrites the active region).
    integer init_i;
    initial for (init_i=0; init_i<DEPTH; init_i=init_i+1) begin
        L1[init_i] = 8'd0; L2[init_i] = 8'd0;
    end

    // 3-tap column shifts. c=current row, a=row y-1, b=row y-2.
    // Tap 0 is the newest column (x), tap 2 is x-2.
    reg [7:0] c0,c1,c2;
    reg [7:0] a0,a1,a2;
    reg [7:0] b0,b1,b2;

    // rgb/gray/vs/de delayed 2 stages; combined with the output register this
    // yields the same 3-cycle latency as the Sobel window.
    reg [23:0] rgb_d1, rgb_d2;
    reg [7:0]  gray_d1, gray_d2;
    reg vs_d1, vs_d2, de_d1, de_d2;

    // Causal window centred at (x-1, y-1):
    //   top    b2 b1 b0      middle a2 a1 a0      bottom c2 c1 c0
    wire signed [10:0] gx = $signed({3'b0,b0}) + $signed({2'b0,a0,1'b0}) + $signed({3'b0,c0})
                          - $signed({3'b0,b2}) - $signed({2'b0,a2,1'b0}) - $signed({3'b0,c2});
    wire signed [10:0] gy = $signed({3'b0,c2}) + $signed({2'b0,c1,1'b0}) + $signed({3'b0,c0})
                          - $signed({3'b0,b2}) - $signed({2'b0,b1,1'b0}) - $signed({3'b0,b0});
    wire [10:0] mag = (gx<0 ? -gx : gx) + (gy<0 ? -gy : gy);
    wire edge_detected = (mag > {3'b0, EDGE_TH});

    // Thermal: piecewise linear false color on gray_d2 (blue->cyan->green->yellow->red).
    reg [7:0] t_r, t_g, t_b;
    always @* begin
        if      (gray_d2 < 8'd64)  begin t_r = 8'd0;                 t_g = gray_d2<<2;         t_b = 8'hFF; end
        else if (gray_d2 < 8'd128) begin t_r = 8'd0;                 t_g = 8'hFF;             t_b = 8'hFF - ((gray_d2-8'd64)<<2); end
        else if (gray_d2 < 8'd192) begin t_r = (gray_d2-8'd128)<<2;  t_g = 8'hFF;             t_b = 8'd0; end
        else                       begin t_r = 8'hFF;                t_g = 8'hFF - ((gray_d2-8'd192)<<2); t_b = 8'd0; end
    end

    always @(posedge clk or posedge rst) begin
        if (rst) begin
            c0<=0;c1<=0;c2<=0; a0<=0;a1<=0;a2<=0; b0<=0;b1<=0;b2<=0;
            rgb_d1<=0;rgb_d2<=0; gray_d1<=0;gray_d2<=0;
            vs_d1<=0;vs_d2<=0; de_d1<=0;de_d2<=0;
            out_vs<=0; out_de<=0; out_rgb<=0;
        end else begin
            c2<=c1; c1<=c0; c0<=gray;
            a2<=a1; a1<=a0; a0<=l1;
            b2<=b1; b1<=b0; b0<=l2;
            rgb_d1<=rgb;  rgb_d2<=rgb_d1;
            gray_d1<=gray; gray_d2<=gray_d1;
            vs_d1<=vs; vs_d2<=vs_d1; de_d1<=de; de_d2<=de_d1;
            out_vs<=vs_d2; out_de<=de_d2;

            // Shift rows during active pixels: old L1 -> L2, current gray -> L1.
            // Blanking performs no write, so the last full row is preserved.
            if (de) begin
                L2[x] <= l1;
                L1[x] <= gray;
            end

            case (mode_sel)
                2'b00: out_rgb <= rgb_d2;
                2'b01: out_rgb <= {8'hFF-rgb_d2[23:16], 8'hFF-rgb_d2[15:8], 8'hFF-rgb_d2[7:0]};
                2'b10: out_rgb <= edge_detected ? 24'hFFFFFF : 24'h000000;
                2'b11: out_rgb <= {t_r, t_g, t_b};
                default: out_rgb <= rgb_d2;
            endcase
        end
    end
endmodule
