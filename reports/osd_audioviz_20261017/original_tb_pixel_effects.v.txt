`timescale 1ns/1ps
module tb_pixel_effects;
 reg clk=0,rst=1,vs=0,de=0; always #5 clk=~clk;
 reg [9:0] x=0; reg [23:0] rgb=0; reg [1:0] mode=0;
 wire ov,od; wire [23:0] color;
 integer errors=0, i, yy, xx, idx, oidx;

 pixel_effects #(.DEPTH(1024), .EDGE_TH(8'd80)) dut(clk,rst,vs,de,x,rgb,mode,ov,od,color);

 reg [23:0] exp_lin [0:4095];
 reg [23:0] out_lin  [0:4095];

 // Capture active outputs gated by out_de (== de delayed 3, aligned with out_rgb).
 // Sample at negedge so the module's posedge NBA has already settled.
 always @(negedge clk) begin
   if (rst) oidx = 0;
   else if (od) begin out_lin[oidx] = color; oidx = oidx + 1; end
 end

 // Golden model for the 4 modes, evaluated inline so there is no delta race.
 function [23:0] expected; input [23:0] p; input [1:0] m;
   reg [7:0] g;
 begin
   case (m)
     2'b01: expected = {8'hFF-p[23:16], 8'hFF-p[15:8], 8'hFF-p[7:0]};
     2'b11: begin
       g = (77*p[23:16] + 150*p[15:8] + 29*p[7:0]) >> 8;
       if      (g < 8'd64)  expected = {8'd0, g<<2, 8'hFF};
       else if (g < 8'd128) expected = {8'd0, 8'hFF, 8'hFF-((g-8'd64)<<2)};
       else if (g < 8'd192) expected = {(g-8'd128)<<2, 8'hFF, 8'd0};
       else                 expected = {8'hFF, 8'hFF-((g-8'd192)<<2), 8'd0};
     end
     default: expected = p;
   endcase
 end
 endfunction

 // ---- Sobel golden reference: causal window centred at (ox+1, oy-1) ----
 reg [7:0] fr [0:7][0:7];
 function [7:0] fg; input integer r,c;
 begin
   if (r<0 || r>=8 || c<0 || c>=8) fg = 8'd0; else fg = fr[r][c];
 end
 endfunction
 function integer sobel_mag; input integer oy,ox;
   integer cy,cx,gx,gy,a,b;
 begin
   cy = oy-1; cx = ox+1;
   a = fg(cy-1,cx+1) + 2*fg(cy,cx+1) + fg(cy+1,cx+1);
   b = fg(cy-1,cx-1) + 2*fg(cy,cx-1) + fg(cy+1,cx-1);
   gx = a - b;
   a = fg(cy+1,cx-1) + 2*fg(cy+1,cx) + fg(cy+1,cx+1);
   b = fg(cy-1,cx-1) + 2*fg(cy-1,cx) + fg(cy-1,cx+1);
   gy = a - b;
   if (gx<0) gx = -gx;
   if (gy<0) gy = -gy;
   sobel_mag = gx + gy;
 end
 endfunction

 task do_reset; begin
   rst=1; vs=0; de=0; x=0; rgb=0; mode=0;
   repeat(3) @(posedge clk);
   rst=0;
   repeat(2) @(posedge clk);
 end
 endtask

 initial begin
   // ---------- passthrough ----------
   do_reset; mode=2'b00; idx=0;
   for (i=0;i<256;i=i+1) begin
     x=i[9:0]; de=1; rgb={i[7:0], i[6:0], i[7:1]};
     exp_lin[idx]=expected(rgb,mode); idx=idx+1;
     @(posedge clk);
   end
   de=0; repeat(6) @(posedge clk);
   if (oidx!=idx) begin $display("PASSTHROUGH count %0d != %0d", oidx, idx); errors=errors+1; end
   else for (i=0;i<idx;i=i+1) if (out_lin[i]!==exp_lin[i]) begin
     if (errors<8) $display("PASSTHROUGH[%0d] got=%h want=%h", i, out_lin[i], exp_lin[i]);
     errors=errors+1;
   end

   // ---------- invert ----------
   do_reset; mode=2'b01; idx=0;
   for (i=0;i<256;i=i+1) begin
     x=i[9:0]; de=1; rgb={i[7:0], 8'hFF-i[7:0], i[7:1]};
     exp_lin[idx]=expected(rgb,mode); idx=idx+1;
     @(posedge clk);
   end
   de=0; repeat(6) @(posedge clk);
   if (oidx!=idx) begin $display("INVERT count %0d != %0d", oidx, idx); errors=errors+1; end
   else for (i=0;i<idx;i=i+1) if (out_lin[i]!==exp_lin[i]) begin
     if (errors<8) $display("INVERT[%0d] got=%h want=%h", i, out_lin[i], exp_lin[i]);
     errors=errors+1;
   end

   // ---------- thermal (exhaustive 0..255) ----------
   do_reset; mode=2'b11; idx=0;
   for (i=0;i<256;i=i+1) begin
     x=i[9:0]; de=1; rgb={i[7:0], i[7:0], i[7:0]};   // gray == i
     exp_lin[idx]=expected(rgb,mode); idx=idx+1;
     @(posedge clk);
   end
   de=0; repeat(6) @(posedge clk);
   if (oidx!=idx) begin $display("THERMAL count %0d != %0d", oidx, idx); errors=errors+1; end
   else for (i=0;i<idx;i=i+1) if (out_lin[i]!==exp_lin[i]) begin
     if (errors<8) $display("THERMAL[%0d] got=%h want=%h", i, out_lin[i], exp_lin[i]);
     errors=errors+1;
   end

   // ---------- Sobel (8x8 quadrant, full-frame golden) ----------
   for (yy=0;yy<8;yy=yy+1) for (xx=0;xx<8;xx=xx+1)
     fr[yy][xx] = (xx>=4 && yy>=4) ? 8'd200 : 8'd40;
   do_reset; mode=2'b10; idx=0;
   // 16 cycles/line: 8 active (x=0..7, de=1) then 8 blanking (x=8..15, de=0, rgb=0)
   // so line-buffer reads during blanking hit unwritten entries == zero-padding.
   for (yy=0;yy<8;yy=yy+1) begin
     for (xx=0;xx<16;xx=xx+1) begin
       x=xx[9:0]; de=(xx<8);
       rgb=(xx<8)?{fr[yy][xx],fr[yy][xx],fr[yy][xx]}:24'h0;
       if (xx<8) begin
         exp_lin[idx] = (sobel_mag(yy,xx) > 80) ? 24'hFFFFFF : 24'h000000;
         idx=idx+1;
       end
       @(posedge clk);
     end
   end
   repeat(6) @(posedge clk);
   if (oidx!=idx) begin $display("SOBEL count %0d != %0d", oidx, idx); errors=errors+1; end
   else for (i=0;i<idx;i=i+1) if (out_lin[i]!==exp_lin[i]) begin
     if (errors<64) $display("SOBEL[%0d] y=%0d x=%0d got=%h want=%h", i, i/8, i%8, out_lin[i], exp_lin[i]);
     errors=errors+1;
   end

   // ---------- vs alignment (same 3-stage path as de) ----------
   do_reset; vs=0; de=0; repeat(2) @(posedge clk);
   vs=1; @(posedge clk); vs=0;
   @(posedge clk); @(posedge clk); #1;
   if (ov!==1'b1) begin $display("VS high fail ov=%b", ov); errors=errors+1; end
   @(posedge clk); #1;
   if (ov!==1'b0) begin $display("VS low fail ov=%b", ov); errors=errors+1; end

   if (errors==0) $display("ALL PASS tb_pixel_effects: passthrough/invert/thermal exact, Sobel 8x8 golden, vs/de alignment");
   else $display("FAILED tb_pixel_effects %0d", errors);
   $finish;
 end

 initial begin #2000000; $display("FAILED watchdog tb_pixel_effects"); $finish; end
endmodule
